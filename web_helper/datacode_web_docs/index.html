Что такое DataCode?

DataCode — это простой интерактивный язык программирования, разработанный для быстрой обработки данных и легкого обучения. Он включает интуитивный синтаксис, мощную поддержку массивов, встроенные функции и пользовательские функции с локальной областью видимости.

Простой пример

DataCode 
# Простой пример DataCode
let arr = [1, 2, 3, 4, 5]

for item in arr {
    print("Number:", item, "Squared:", item * item)
}

# Результат:
# Number: 1 Squared: 1
# Number: 2 Squared: 4
# Number: 3 Squared: 9
# Number: 4 Squared: 16
# Number: 5 Squared: 25

Установка

Вариант 1: Глобальная установка (Рекомендуется)

Установите DataCode как глобальную команду:

Bash 
# Клонируйте и установите
git clone https://github.com/igornet0/DataCode.git
cd DataCode

# Установите глобально
make install
# или
./install.sh

# Теперь вы можете использовать datacode откуда угодно!
datacode --help

Вариант 2: Режим разработки

Запускайте напрямую с помощью Cargo:

Bash 
git clone https://github.com/igornet0/DataCode.git
cd DataCode
cargo run

Использование после установки

Bash 
datacode                   # Запустить интерактивный REPL
datacode filename.dc       # Выполнить файл DataCode
datacode filename.dc --build_model  # Выполнить и экспортировать таблицы в SQLite
datacode filename.dc --build_model output.db  # Экспортировать в указанный файл
datacode --repl            # Запустить интерактивный REPL
datacode --help            # Показать справку

Рекурсивные функции

DataCode 
# Факториал через рекурсию
fn factorial(n) {
    if n <= 1 {
        return 1
    }
    return n * factorial(n - 1)
}

# Числа Фибоначчи
fn fibonacci(n) {
    if n <= 1 {
        return n
    }
    return fibonacci(n - 1) + fibonacci(n - 2)
}

# Сумма чисел от 1 до n
fn sum_to_n(n) {
    if n <= 0 {
        return 0
    }
    return n + sum_to_n(n - 1)
}

# Тестируем функции
print("Factorial of 5 =", factorial(5))
print("Factorial of 3 =", factorial(3))
print("fibonacci(6) =", fibonacci(6))
print("fibonacci(8) =", fibonacci(8))
print("sum_to_n(10) =", sum_to_n(10))

Создание и обучение нейронной сети

DataCode 
import ml

# Загрузка датасета MNIST
global x_train, y_train, x_test, y_test = ml.dataset.load_mnist()

# Создание нейронной сети
global layer1 = ml.layer.linear(784, 128)
global layer2 = ml.layer.relu()
global layer3 = ml.layer.linear(128, 10)

# Обучение модели

# Предсказание

# Сохранение модели
model.save(path('mnist_model.nn'))

JOIN операции

Объединение таблиц пользователей и заказов

DataCode 
# Создание таблиц
global users = table([[1, 'Alice'], [2, 'Bob'], [3, 'Charlie']], ['id', 'name'])
global orders = table([[1, 100], [1, 200], [3, 300]], ['user_id', 'amount'])

# INNER JOIN - только пользователи с заказами
global users_with_orders = inner_join(users, orders, 'id', 'user_id')

# LEFT JOIN - все пользователи, включая без заказов
global all_users_orders = left_join(users, orders, 'id', 'user_id')

# ANTI JOIN - пользователи без заказов
global users_without_orders = anti_join(users, orders, 'id', 'user_id')

show_table(users_with_orders)
show_table(all_users_orders)
show_table(users_without_orders)

ML модуль импортируется в начале программы:

JOIN операции

Операции объединения таблиц предназначены для объединения двух таблиц на основе равенства ключей, произвольных логических условий, временной близости значений или позиции строк.

Типы JOIN

Тип Описание Функция
inner Возвращает строки, для которых существует совпадение в обеих таблицах inner_join()
left Все строки из left сохраняются, отсутствующие значения из right заполняются NULL left_join()
right Все строки из right сохраняются, отсутствующие значения из left заполняются NULL right_join()
full Возвращает все строки из обеих таблиц, отсутствующие значения заполняются NULL full_join()
cross Декартово произведение таблиц cross_join()
semi Возвращает только строки left, имеющие совпадения в right, без колонок right semi_join()
anti Возвращает строки left, не имеющие совпадений в right anti_join()

Универсальная функция JOIN

DataCode 
join(
    left: Table,
    right: Table,
    on: JoinKeys | Expr,
    type: JoinType = "inner",
    suffixes: (string, string) = ("_left", "_right"),
    nulls_equal: boolean = false
) -> Table

Примеры использования

INNER JOIN

DataCode 
global users = table([[1, 'Alice'], [2, 'Bob']], ['id', 'name'])
global orders = table([[1, 100], [1, 200]], ['user_id', 'amount'])
global result = inner_join(users, orders, 'id', 'user_id')

LEFT JOIN

DataCode 
global users = table([[1, 'Alice'], [2, 'Bob']], ['id', 'name'])
global orders = table([[1, 100]], ['user_id', 'amount'])
global result = left_join(users, orders, 'id', 'user_id')
# Bob будет иметь NULL в колонках orders

CROSS JOIN

DataCode 
global table1 = table([[1], [2]], ['col1'])
global table2 = table([['a'], ['b']], ['col2'])
global result = cross_join(table1, table2)
# Результат: 4 строки (2 * 2)

SEMI JOIN

DataCode 
global users = table([[1, 'Alice'], [2, 'Bob'], [3, 'Charlie']], ['id', 'name'])
global orders = table([[1, 100], [3, 300]], ['user_id', 'amount'])
global result = semi_join(users, orders, 'id', 'user_id')
# Результат: только Alice и Charlie (без колонок orders)

ANTI JOIN

DataCode 
global users = table([[1, 'Alice'], [2, 'Bob'], [3, 'Charlie']], ['id', 'name'])
global orders = table([[1, 100], [3, 300]], ['user_id', 'amount'])
global result = anti_join(users, orders, 'id', 'user_id')
# Результат: только Bob (без заказов)

ASOF JOIN (временное соединение)

DataCode 
global prices = table([['2024-01-01', 100], ['2024-01-05', 105]], ['date', 'price'])
global trades = table([['2024-01-02', 'buy'], ['2024-01-06', 'sell']], ['date', 'action'])
global result = asof_join(trades, prices, 'date', 'date', 'backward')

WebSocket сервер

WebSocket сервер для удаленного выполнения кода DataCode.

Запуск сервера

Bash 
# Запуск на адресе по умолчанию (127.0.0.1:8080)
datacode --websocket

# Запуск с указанием хоста и порта через флаги
datacode --websocket --host 0.0.0.0 --port 8899

# Запуск на кастомном адресе через переменную окружения
DATACODE_WS_ADDRESS=0.0.0.0:3000 datacode --websocket

Протокол

Подключитесь к WebSocket серверу по адресу ws://127.0.0.1:8080 (или указанному адресу).

Выполнение кода

Отправьте JSON сообщение с типом execute и полем code:

JSON 
{
  "type": "execute",
  "code": "print('Hello, World!')"
}

Подключение к SMB шаре

Для подключения к SMB (Samba/CIFS) шаре используйте тип smb_connect:

JSON 
{
  "type": "smb_connect",
  "ip": "192.168.1.100",
  "login": "username",
  "password": "password",
  "domain": "WORKGROUP",
  "share_name": "share_name"
}

Формат ответа

Успешное выполнение:

JSON 
{
  "success": true,
  "output": "Hello, World!\n",
  "error": null
}

Ошибка выполнения:

JSON 
{
  "success": false,
  "output": "",
  "error": "Error message here"
}

Примеры использования

JavaScript

JavaScript 
const ws = new WebSocket('ws://127.0.0.1:8080');

ws.onopen = () => {
    ws.send(JSON.stringify({
        type: 'execute',
        code: 'print("Hello from JavaScript!")'
    }));
};

ws.onmessage = (event) => {
    const response = JSON.parse(event.data);
    console.log('Output:', response.output);
    if (response.error) {
        console.error('Error:', response.error);
    }
};

Python

Python 
import websocket
import json

def on_message(ws, message):
    response = json.loads(message)
    print('Output:', response['output'])
    if response['error']:
        print('Error:', response['error'])

ws = websocket.WebSocketApp('ws://127.0.0.1:8080',
                            on_message=on_message)
ws.on_open = lambda ws: ws.send(json.dumps({
    'type': 'execute',
    'code': 'print("Hello from Python!")'
}))
ws.run_forever()

Экспорт в SQLite (--build_model)

DataCode поддерживает автоматический экспорт всех таблиц из глобальных переменных в базу данных SQLite с автоматическим определением зависимостей между таблицами.

Основные возможности:

  • ✅ Автоматический экспорт таблиц — все таблицы из глобальных переменных экспортируются в отдельные таблицы SQLite
  • ✅ Метаданные переменных — создается таблица _datacode_variables с информацией о всех глобальных переменных
  • ✅ Автоматическое определение зависимостей — система автоматически находит связи между таблицами по ID столбцам
  • ✅ Создание индексов — автоматическое создание индексов для ID столбцов и внешних ключей
  • ✅ Преобразование типов — автоматическое преобразование типов DataCode в типы SQLite

Пример использования:

DataCode 
# Загрузка данных
global sales = read("sales.csv")
global products = read("products.csv")
global customers = read("customers.csv")

# Обработка данных
global sales_table = table(sales)
global products_table = table(products)
global customers_table = table(customers)

# Фильтрация и преобразование
global filtered_sales = table_where(sales_table, "amount > 100")

datacode load_model_data.dc --build_model
